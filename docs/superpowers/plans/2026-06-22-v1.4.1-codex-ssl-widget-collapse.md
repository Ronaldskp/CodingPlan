# V1.4.1 Codex SSL/Network Resilience + Widget Collapse/Refresh Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans or superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Harden Codex network error handling, preserve last-known quota values on provider failures, add a clickable collapse/expand summary to the floating widget, and add an in-window manual refresh button without changing existing CLI, tray, or EXE behavior.

**Architecture:** Keep provider/core/ui layering. Codex provider translates SSL/timeout/proxy exceptions into short UI-safe messages. `core/collector` merges newly failed provider results with the previous cached values so the UI never blanks out known data. `core/ui_state` persists the new `collapsed` flag. `ui/widget` adds a header bar with refresh/collapse buttons, a one-line collapsed summary, and header-click/drag handling. Doctor and docs are updated with Codex troubleshooting guidance.

**Tech Stack:** Python 3.10+, PySide6, urllib, pytest, PyInstaller onedir.

import ssl

def _short_network_message(exc: Exception) -> str:
    text = sanitize_message(exc, max_length=200)
    lower = text.lower()
    if "ssl" in lower:
        return "SSL/network error"
    if "proxy" in lower or "tunnel" in lower:
        return "Proxy/network error"
    if "time" in lower or "timeout" in lower:
        return "Network timeout"
    return "Network error"
```

def _preserve_stale_values(
    old_usage: dict[str, Any], new_usage: dict[str, Any]
) -> dict[str, Any]:
    old_providers = old_usage.get("providers", {}) if isinstance(old_usage, dict) else {}
    new_providers = new_usage.get("providers", {})
    for key, new_provider in new_providers.items():
        if new_provider.get("status") == "ok":
            continue
        old_provider = old_providers.get(key)
        if not isinstance(old_provider, dict):
            continue
        fallback = _provider_fallback_values(old_provider)
        if fallback:
            debug = sanitize_message(new_provider.get("message", ""))
            new_provider.update(fallback)
            new_provider["debug"] = debug
            new_provider["message"] = "Last value shown; refresh failed."
    return new_usage
```

def _provider_fallback_values(old_provider: dict[str, Any]) -> dict[str, Any] | None:

def read_ui_state(path: Path) -> dict[str, Any]:
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError, UnicodeDecodeError):
        return {"collapsed": False}
    if not isinstance(data, dict):
        return {"collapsed": False}
    state: dict[str, Any] = {"collapsed": False}
    if isinstance(data.get("x"), int):
        state["x"] = data["x"]
    if isinstance(data.get("y"), int):
        state["y"] = data["y"]
    if isinstance(data.get("always_on_top"), bool):
        state["always_on_top"] = data["always_on_top"]
    if isinstance(data.get("collapsed"), bool):
        state["collapsed"] = data["collapsed"]
    for key, value in data.items():
        if key not in state:
            state[key] = value
    return state


def write_ui_state(path: Path, state: dict[str, Any]) -> None:
    existing = read_ui_state(path) if path.exists() else {}
    clean: dict[str, Any] = {**existing}

    clean["y"] = int(state["y"])

    collapsed = state.get("collapsed")

        clean["collapsed"] = collapsed
    elif "collapsed" not in clean:
        clean["collapsed"] = False
    for key, value in state.items():
        if key not in clean:
            clean[key] = value

```python
def _toggle_collapse(self) -> None:
    self.collapsed = not self.collapsed
    self._apply_collapse_state()
    self._save_state()


def _apply_collapse_state(self, initial: bool = False) -> None:
    width = int(self.config["ui"].get("width", 280))
    if self.collapsed:
        self.cards_widget.hide()
        self.summary.show()
        self.collapse_btn.setText("˅")
        self.setFixedHeight(self.collapsed_height)
        if not initial:
            self.resize(width, self.collapsed_height)
    else:
        self.cards_widget.show()
        self.summary.hide()
        self.collapse_btn.setText("˄")
        self.setMaximumSize(16777215, 16777215)
        self.setMinimumSize(0, 0)
        if not initial:
            self.resize(width, self.expanded_height)
    self._update_summary()


def _update_summary(self) -> None:
    usage = read_usage(self.cache_path, create_if_missing=True)
    providers = usage.get("providers", {})
    parts = []
    codex = providers.get("codex", {})
    if codex.get("status") == "ok":
        parts.append(f"C {_pct(codex.get('fiveHour'))}/{_pct(codex.get('weekly'))}")
    else:
        parts.append("C offline")

    kimi = providers.get("kimi", {})
    if kimi.get("status") == "ok":
        parts.append(f"K {_pct(kimi.get('fiveHour'))}/{_pct(kimi.get('weekly'))}")
    else:
        parts.append("K offline")

    deepseek = providers.get("deepseek", {})
    balance = deepseek.get("balance") or {}
    if deepseek.get("status") == "ok" and balance.get("total") is not None:
        parts.append(f"DS {float(balance['total']):.2f}")
    else:
        parts.append("DS offline")

    self.summary.setText("  ".join(parts))


def _pct(window: Any) -> str:
    if isinstance(window, dict) and window.get("remainingPercent") is not None:
        return f"{round(window['remainingPercent'])}"
    return "--"
```

```python
def mousePressEvent(self, event: QMouseEvent) -> None:
    if event.button() == Qt.MouseButton.LeftButton:
        self._drag_start = event.globalPosition().toPoint()
        self.drag_offset = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
        self._is_dragging = False
        event.accept()


def mouseMoveEvent(self, event: QMouseEvent) -> None:
    if self.drag_offset is not None and event.buttons() & Qt.MouseButton.LeftButton:
        moved = (event.globalPosition().toPoint() - self._drag_start).manhattanLength()
        if moved > self._drag_threshold:
            self._is_dragging = True
        self.move(event.globalPosition().toPoint() - self.drag_offset)
        event.accept()


def mouseReleaseEvent(self, event: QMouseEvent) -> None:
    self.drag_offset = None
    if (
        not self._is_dragging
        and event.button() == Qt.MouseButton.LeftButton
        and self.click_title_to_toggle
        and self.header_widget is not None
        and self.header_widget.geometry().contains(event.pos())
    ):
        self._toggle_collapse()
    self._is_dragging = False
```
