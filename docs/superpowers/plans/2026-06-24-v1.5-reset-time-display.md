```python
"""Reset time parsing and display helpers."""

from __future__ import annotations

import re
from datetime import datetime, timedelta, timezone
from typing import Any


_TIME_ONLY_RE = re.compile(r"^\d{1,2}:\d{2}$")
_SHORT_RE = re.compile(r"^(\d{2})-(\d{2})\s+(\d{1,2}):(\d{2})$")


def _try_parse_iso(value: str) -> datetime | None:
    try:
        parsed = datetime.fromisoformat(value)
        if parsed.tzinfo is None:
            parsed = parsed.replace(tzinfo=timezone(timedelta()))
        return parsed
    except (TypeError, ValueError):
        return None


def _try_parse_short(value: str, now: datetime) -> datetime | None:
    match = _SHORT_RE.match(value)
    if not match:
        return None
    month, day, hour, minute = (int(g) for g in match.groups())
    candidate = now.replace(month=month, day=day, hour=hour, minute=minute, second=0, microsecond=0)
    if candidate < now - timedelta(days=180):
        candidate = candidate.replace(year=candidate.year + 1)
    return candidate


def _try_parse_time_only(value: str, now: datetime) -> datetime | None:
    if not _TIME_ONLY_RE.match(value):
        return None
    hour, minute = (int(part) for part in value.split(":"))
    candidate = now.replace(hour=hour, minute=minute, second=0, microsecond=0)
    if candidate < now - timedelta(hours=12):
        candidate += timedelta(days=1)
    return candidate


def parse_reset_time(value: Any, now: datetime | None = None) -> datetime | None:
    if value is None or value == "":
        return None
    text = str(value).strip()
    current = now or datetime.now().astimezone()
    for parser in (_try_parse_iso, _try_parse_short, _try_parse_time_only):
        parsed = parser(text, current) if parser.__name__ == "_try_parse_time_only" else (
            parser(text) if parser.__name__ == "_try_parse_iso" else parser(text, current)
        )
        if parsed is not None:
            return parsed
    return None


def format_reset_time(reset_at: Any, now: datetime | None = None) -> str:
    parsed = parse_reset_time(reset_at, now)
    if parsed is None:
        return "--" if reset_at in (None, "") else str(reset_at)
    current = now or datetime.now().astimezone()
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=current.tzinfo)
    if parsed <= current:
        return "soon"
    if parsed.date() == current.date():
        return parsed.strftime("%H:%M")
    return parsed.strftime("%m-%d %H:%M")


def format_reset_countdown(reset_at: Any, now: datetime | None = None) -> str:
    parsed = parse_reset_time(reset_at, now)
    if parsed is None:
        return "--"
    current = now or datetime.now().astimezone()
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=current.tzinfo)
    delta = parsed - current
    if delta.total_seconds() <= 0:
        return "soon"
    hours, remainder = divmod(int(delta.total_seconds()), 3600)
    minutes = remainder // 60
    return f"{hours}h{minutes:02d}m"
```

```python
def _format_compact_summary(providers: dict[str, Any], show_reset: bool = True) -> str:
    codex = providers.get("codex", {})
    kimi = providers.get("kimi", {})
    deepseek = providers.get("deepseek", {})

    def pct(window: Any) -> str:
        if isinstance(window, dict) and window.get("remainingPercent") is not None:
            return f"{round(window['remainingPercent'])}"
        return "--"

    def provider_line(provider: dict[str, Any], key: str) -> str:
        if provider.get("status") != "ok":
            return f"{key} offline"
        five = provider.get("fiveHour", {})
        weekly = provider.get("weekly", {})
        return f"{key} {pct(five)}/{pct(weekly)}"

    parts = [
        f"C {provider_line(codex, '')}".replace("C ", "C "),
        f"K {provider_line(kimi, '')}".replace("K ", "K "),
    ]
    balance = deepseek.get("balance") or {}
    if deepseek.get("status") == "ok" and balance.get("total") is not None:
        parts.append(f"DS {float(balance['total']):.2f}")
    else:
        parts.append("DS offline")
    summary = "  ".join(parts)

    if not show_reset:
        return summary

    reset_parts = []
    for label, provider in (("C", codex), ("K", kimi)):
        if provider.get("status") != "ok":
            continue
        five = provider.get("fiveHour", {})
        if isinstance(five, dict):
            reset = format_reset_time(five.get("resetAt"))
            if reset != "--":
                reset_parts.append(f"{label}{reset}")
    reset_line = f"R {' '.join(reset_parts)}" if reset_parts else ""
    return f"{summary}\n{reset_line}".rstrip("\n")
```

# UI Notes

## Design Principles

- Information first: show percentages, balances, and reset times at a glance.
- Small window: default 280×300 expanded, compact collapsed mode.
- Short error text in cards (`offline`, `network error`, `unsupported`).
- Full provider messages go to debug logs/tooltips, not the small card.

## Reset Time Display

- Codex reset time comes from the official usage response or custom command JSON.
- Kimi reset time comes from `kimi-usage --json` (`5h Limit` / `Weekly Usage` items).
- If reset time is missing or unparsable, the UI shows `--`.
- Stale fallback preserves the last known reset time alongside the last known percentage.

## Collapsed Summary

```text
C 99/75  K 70/52  DS 58.23
R C18:44 K18:44    Fresh
```

- First line: Codex/Kimi 5H/WEEK percentages and DeepSeek balance.
- Second line (optional): 5H reset times and cache freshness.

## Config

```yaml
ui:
  show_reset_times: true
  reset_display_mode: "time"
  compact_summary_show_reset: true
```
