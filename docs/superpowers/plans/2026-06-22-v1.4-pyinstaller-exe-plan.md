```python
"""Runtime path resolution for source and PyInstaller EXE runs."""

from __future__ import annotations

import sys
from pathlib import Path
from typing import Any


def _is_frozen() -> bool:
    return getattr(sys, "frozen", False)


def project_root() -> Path:
    """Return the directory that should contain .env, config.yaml, data/, assets/."""
    if _is_frozen():
        # EXE runs next to its bundled files; do NOT use sys._MEIPASS for writes.
        return Path(sys.executable).resolve().parent
    # Source run: core/paths.py -> core/ -> project root.
    return Path(__file__).resolve().parent.parent


def meipass_root() -> Path | None:
    """PyInstaller's temporary extraction folder, for bundled read-only resources only."""
    meipass = getattr(sys, "_MEIPASS", None)
    return Path(meipass) if meipass else None


def _resolve_external(relative: str | Path) -> Path:
    path = Path(relative)
    if path.is_absolute():
        return path
    return project_root() / path


def data_path(relative: str | Path = "") -> Path:
    base = project_root() / "data"
    if not relative:
        return base
    return base / relative


def ensure_data_dir() -> Path:
    path = data_path()
    path.mkdir(parents=True, exist_ok=True)
    return path


def config_path() -> Path:
    return project_root() / "config.yaml"


def dotenv_path() -> Path:
    return project_root() / ".env"


def assets_path(relative: str | Path = "") -> Path:
    base = project_root() / "assets"
    if not relative:
        return base
    return base / relative


def application_icon_path() -> Path:
    """Prefer external icon, otherwise fall back to a bundled read-only icon."""
    external = assets_path("coding_plan.ico")
    if external.exists():
        return external
    meipass = meipass_root()
    if meipass:
        bundled = meipass / "assets" / "coding_plan.ico"
        if bundled.exists():
            return bundled
    return external  # Return non-existent external path so callers can check exists().


def load_dotenv_if_present(override: bool = False) -> None:
    """Load external .env if it exists."""
    from dotenv import load_dotenv

    path = dotenv_path()
    if path.exists():
        load_dotenv(str(path), override=override)


def resolve_configured_path(configured: str | Path) -> Path:
    """Resolve a configured relative path against the runtime project root."""
    path = Path(configured)
    return path if path.is_absolute() else project_root() / path
```

"""Tests for runtime path resolution."""

from __future__ import annotations

import sys
from pathlib import Path

import pytest

import core.paths as paths


def test_project_root_in_source_mode():
    root = paths.project_root()
    assert (root / "main.py").exists()
    assert (root / "config.yaml").exists()


def test_data_path_creates_parent_on_demand(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    assert paths.data_path() == tmp_path / "data"


def test_ensure_data_dir_creates_directory(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    result = paths.ensure_data_dir()
    assert result.exists()
    assert result.is_dir()


def test_config_path_and_dotenv_path(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    assert paths.config_path() == tmp_path / "config.yaml"
    assert paths.dotenv_path() == tmp_path / ".env"


def test_assets_path_returns_external_path(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    assert paths.assets_path("coding_plan.ico") == tmp_path / "assets" / "coding_plan.ico"


def test_application_icon_path_prefers_external(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    (tmp_path / "assets").mkdir()
    (tmp_path / "assets" / "coding_plan.ico").write_text("fake", encoding="utf-8")
    assert paths.application_icon_path() == tmp_path / "assets" / "coding_plan.ico"


def test_application_icon_path_falls_back_to_meipass(tmp_path, monkeypatch):
    monkeypatch.setattr(paths, "project_root", lambda: tmp_path)
    meipass = tmp_path / "_internal"
    (meipass / "assets").mkdir(parents=True)
    (meipass / "assets" / "coding_plan.ico").write_text("bundled", encoding="utf-8")
    monkeypatch.setattr(sys, "_MEIPASS", str(meipass), raising=False)
    assert paths.application_icon_path() == meipass / "assets" / "coding_plan.ico"

    monkeypatch.delattr(sys, "_MEIPASS", raising=False)
    assert paths.application_icon_path() == tmp_path / "assets" / "coding_plan.ico"
    assert not paths.application_icon_path().exists()


def test_resolve_configured_path_absolute():

"""Safety checks for the release package produced by build_exe.bat."""

from __future__ import annotations

import re
from pathlib import Path

import pytest

RELEASE_ROOT = Path(__file__).resolve().parent.parent / "release" / "CodingPlanWidget"
SENSITIVE_RE = re.compile(
    r"(KIMI(?:_CODING)?_API_KEY|DEEPSEEK_API_KEY|Bearer\s+[A-Za-z0-9._~+/=-]+|"
    r"sk-[A-Za-z0-9_-]{6,}|eyJ[A-Za-z0-9._-]{10,})",
    re.IGNORECASE,
)


def test_release_dotenv_example_exists():
    if not RELEASE_ROOT.exists():
        pytest.skip("release package not built yet")
    assert (RELEASE_ROOT / ".env.example").exists()


def test_release_real_dotenv_missing():
    if not RELEASE_ROOT.exists():
        pytest.skip("release package not built yet")
    assert not (RELEASE_ROOT / ".env").exists()


def test_release_data_directory_empty_of_caches():
    if not RELEASE_ROOT.exists():
        pytest.skip("release package not built yet")
    data_dir = RELEASE_ROOT / "data"
    assert data_dir.exists()
    for forbidden in ("usage.json", "deepseek_daily.json", "ui_state.json"):
        assert not (data_dir / forbidden).exists(), f"{forbidden} should not be shipped"


def test_release_no_auth_json():
    if not RELEASE_ROOT.exists():
        pytest.skip("release package not built yet")
    for path in RELEASE_ROOT.rglob("auth.json"):
        pytest.fail(f"Codex auth.json found in release: {path}")


def test_release_text_files_contain_no_secrets():
    if not RELEASE_ROOT.exists():
        pytest.skip("release package not built yet")
    for path in RELEASE_ROOT.rglob("*"):
        if not path.is_file():
            continue
        if path.suffix.lower() in {".exe", ".dll", ".pyd", ".ico", ".png", ".jpg"}:
            continue
        try:
            text = path.read_text(encoding="utf-8", errors="ignore")
        except OSError:
            continue
        if SENSITIVE_RE.search(text):
            pytest.fail(f"Potential secret found in {path}")
```

```markdown
# Packaging Notes

## Why PyInstaller

PyInstaller bundles the Python interpreter and dependencies into a single folder that behaves like a native Windows application. It requires no changes to the PySide6 code beyond path handling.

## Why onedir instead of onefile

- PySide6 onefile EXEs have slower cold-start because the onefile bootloader extracts everything to a temp folder before importing Qt.
- onedir makes it easier to inspect bundled files and debug missing plugins.
- External `config.yaml`, `.env`, and `data/` are clearer in onedir layout.

## Entry Point

`app_launcher.py` is the PyInstaller entry point. It calls `main(["widget"])`, so the EXE always launches the GUI. The original `main.py` CLI remains available from source.

## External Files

The EXE reads and writes from its own directory, not from `sys._MEIPASS`:

- `.env` — loaded from `CodingPlanWidget.exe` sibling directory.
- `config.yaml` — loaded from the same directory.
- `data/` — created and written there.
- `assets/coding_plan.ico` — preferred external; bundled icon is a fallback.

## Why .env is not embedded

Embedding API keys in a binary does not provide real security and makes key rotation impossible. Keeping `.env` external lets users configure keys without rebuilding.

## Release Layout

```text
release\CodingPlanWidget\
├─ CodingPlanWidget.exe
├─ config.yaml
├─ .env.example
├─ README_RELEASE.md
├─ assets\coding_plan.ico
└─ data\
   ├─ .gitkeep
   └─ README.md
```

## Troubleshooting

### PySide6 plugin missing

If the widget fails with platform plugin errors, add the `platforms` directory to `datas` in the spec.

### Icon not showing

Ensure `assets/coding_plan.ico` exists next to the EXE. If missing, the app falls back to a generated CP icon.

### .env not read

Check that `.env` is in the same folder as `CodingPlanWidget.exe`, not inside `dist/`.

### Cannot write data

The EXE must have write permission to its own `data/` directory. Do not install into `Program Files` without adjusting permissions.

### Antivirus false positive

PyInstaller bootloaders are sometimes flagged. If this happens, add an exclusion or build with a custom signed bootloader.

## Future Options

- onefile mode for a single-file distribution.
- WiX/NSIS installer for `Program Files` deployment.
- Windows Task Scheduler for startup.
```
